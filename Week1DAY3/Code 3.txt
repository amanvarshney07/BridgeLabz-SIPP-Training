BEGIN PatientRecordHashing

CLASS Patient
    STRING patientID
    STRING name
    INTEGER age
    STRING diagnosis
ENDCLASS

CLASS PatientRecordManager
    MAP<String, Patient> patientMap

    FUNCTION PatientRecordManager()
        CREATE empty MAP<String, Patient> patientMap
    ENDFUNCTION

    FUNCTION addPatient(Patient p)
        SET patientMap[p.patientID] = p
    ENDFUNCTION

    FUNCTION getPatient(String patientID) RETURNS Patient
        RETURN patientMap[patientID]
    ENDFUNCTION

    FUNCTION updatePatient(String patientID, Patient updatedPatient)
        IF patientMap contains patientID THEN
            SET patientMap[patientID] = updatedPatient
        ENDIF
    ENDFUNCTION

    FUNCTION deletePatient(String patientID)
        REMOVE patientMap[patientID]
    ENDFUNCTION
ENDCLASS
