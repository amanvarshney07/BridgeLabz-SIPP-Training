BEGIN LibraryBookFilter

CLASS Book
    STRING title
    STRING author
    INTEGER yearPublished
    DOUBLE rating
ENDCLASS

CREATE LIST<Book> books

ADD Book("Book1", "Author1", 2016, 4.6) TO books
ADD Book("Book2", "Author2", 2014, 4.8) TO books
ADD Book("Book3", "Author3", 2018, 4.7) TO books
ADD Book("Book4", "Author4", 2017, 4.2) TO books

LIST<Book> filteredBooks = books.stream()
                                .filter(book -> book.yearPublished > 2015)
                                .filter(book -> book.rating > 4.5)
                                .collect(toList)

FOR EACH book IN filteredBooks
    PRINT book.title + " | " + book.author + " | " + book.yearPublished + " | " + book.rating
ENDFOR

END
