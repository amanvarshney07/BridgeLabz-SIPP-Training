BEGIN FlightPriceAnalyzer

CLASS Flight
    STRING flightNumber
    DOUBLE price
    INTEGER duration  // in minutes
ENDCLASS

FUNCTION mergeSort(LIST<Flight> flights) RETURNS LIST<Flight>
    IF flights.size <= 1 THEN
        RETURN flights
    ENDIF

    INTEGER mid = flights.size / 2
    LIST<Flight> left = flights[0 to mid-1]
    LIST<Flight> right = flights[mid to end]

    left = mergeSort(left)
    right = mergeSort(right)

    RETURN merge(left, right)
ENDFUNCTION

FUNCTION merge(LIST<Flight> left, LIST<Flight> right) RETURNS LIST<Flight>
    CREATE empty LIST<Flight> result
    WHILE left not empty AND right not empty
        IF compareFlights(left[0], right[0]) <= 0 THEN
            ADD left[0] TO result
            REMOVE left[0] FROM left
        ELSE
            ADD right[0] TO result
            REMOVE right[0] FROM right
        ENDIF
    ENDWHILE
    ADD remaining elements of left TO result
    ADD remaining elements of right TO result
    RETURN result
ENDFUNCTION

FUNCTION compareFlights(Flight f1, Flight f2) RETURNS INTEGER
    IF f1.price < f2.price THEN
        RETURN -1
    ELSE IF f1.price > f2.price THEN
        RETURN 1
    ELSE
        IF f1.duration < f2.duration THEN
            RETURN -1
        ELSE IF f1.duration > f2.duration THEN
            RETURN 1
        ELSE
            RETURN 0
        ENDIF
    ENDIF
ENDFUNCTION

CREATE LIST<Flight> flights
ADD Flight("F101", 5000, 180) TO flights
ADD Flight("F102", 4500, 200) TO flights
ADD Flight("F103", 5000, 150) TO flights

LIST<Flight> sortedFlights = mergeSort(flights)

FOR EACH flight IN sortedFlights
    PRINT flight.flightNumber + " | " + flight.price + " | " + flight.duration
ENDFOR

END
