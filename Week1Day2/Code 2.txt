BEGIN SmartFarmingSystem

ABSTRACT CLASS Crop
    FUNCTION waterRequirement() RETURNS INTEGER
    FUNCTION harvest() RETURNS STRING
ENDCLASS

CLASS Rice EXTENDS Crop
    FUNCTION waterRequirement()
        RETURN 1000
    ENDFUNCTION

    FUNCTION harvest()
        RETURN "Rice harvested"
    ENDFUNCTION
ENDCLASS

CLASS Wheat EXTENDS Crop
    FUNCTION waterRequirement()
        RETURN 500
    ENDFUNCTION

    FUNCTION harvest()
        RETURN "Wheat harvested"
    ENDFUNCTION
ENDCLASS

CREATE Crop rice = NEW Rice()
CREATE Crop wheat = NEW Wheat()

PRINT rice.waterRequirement()
PRINT rice.harvest()

PRINT wheat.waterRequirement()
PRINT wheat.harvest()

END
