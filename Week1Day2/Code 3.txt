BEGIN OnlineQuizTimer

CLASS QuizTimer
    INTEGER[] timeSlots
    INTEGER currentIndex
    INTEGER totalQuestions

    FUNCTION QuizTimer(totalQuestions)
        SET this.totalQuestions = totalQuestions
        CREATE timeSlots[totalQuestions]
        FOR i = 0 TO totalQuestions-1
            timeSlots[i] = 30  // seconds per question
        ENDFOR
        SET currentIndex = 0
    ENDFUNCTION

    FUNCTION startTimer()
        WHILE true
            DECREMENT timeSlots[currentIndex] by 1
            IF timeSlots[currentIndex] <= 5 THEN
                PRINT "Warning: Time running out for Question " + (currentIndex+1)
            ENDIF

            IF timeSlots[currentIndex] == 0 THEN
                PRINT "Time over for Question " + (currentIndex+1)
                currentIndex = (currentIndex + 1) MOD totalQuestions
            ENDIF

            WAIT 1 second
        ENDWHILE
    ENDFUNCTION
ENDCLASS

CREATE QuizTimer quiz = NEW QuizTimer(5)
CALL quiz.startTimer()

END
